207 double discharge(double current_a, double seconds) {
208 if (current_a < 0)
209 throw std::invalid_argument("Discharge current must be positive");
210 if (is_depleted()) return 0.0;
211
212 double hours = seconds / 3600.0;
213 double ah_requested = current_a * hours;
214 double ah_actual = std::min(ah_requested, remaining_ah_);
215
216 // Average voltage during this discharge step
217 double soc_before = soc();
218 remaining_ah_ -= ah_actual;
219 double soc_after = soc();
220 double avg_soc = (soc_before + soc_after) / 2.0;
221 double avg_voltage = min_voltage_ + avg_soc * (max_voltage_ - min_voltage_)
222 - current_a * internal_resistance_;
223 avg_voltage = std::max(avg_voltage, 0.0);
224
225 return ah_actual * avg_voltage; // Wh
226 }

234 double charge(double current_a, double seconds) {
235 if (current_a < 0)
236 throw std::invalid_argument("Charge current must be positive");
237 if (is_fully_charged()) return 0.0;
238
239 double hours = seconds / 3600.0;
240 double ah_requested = current_a * hours;
241 double ah_space = capacity_ah_ - remaining_ah_;
242 double ah_actual = std::min(ah_requested, ah_space);
243
244 double soc_before = soc();
245 remaining_ah_ += ah_actual;
246 double soc_after = soc();
247 double avg_soc = (soc_before + soc_after) / 2.0;
248 double avg_voltage = min_voltage_ + avg_soc * (max_voltage_ - min_voltage_)
249 + current_a * internal_resistance_;
250
251 if (soc_after >= 0.9999 && soc_before < 0.9999) {
252 ++cycle_count_;
253 }
254
255 return ah_actual * avg_voltage; // Wh
256 }
