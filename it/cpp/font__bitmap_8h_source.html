10// Lightweight placeholder ASCII bitmap generator.
11// Returns an 8x8 monochrome bitmap (8 bytes, each bit is a pixel).
12// This implementation uses a simple deterministic pattern as a
13// placeholder font. Replace entries with a true font bitmap table
14// if you need authentic glyph shapes.
15inline std::array<uint8_t,8> char_to_8x8_bitmap(char c) {
16 std::array<uint8_t,8> out{};
17 uint8_t v = static_cast<uint8_t>(c);
18 for (int r = 0; r < 8; ++r) {
19 // spread character code bits across rows for a reproducible placeholder
20 out[r] = static_cast<uint8_t>(((v << (r%5)) | (v >> (r%3))) & 0xFF);
21 }
22 return out;
23}
