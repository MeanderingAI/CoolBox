122 std::vector<VAEEpochResult> fit(const MatrixD& data, int epochs = 100,
123 int batch_size = 0) {
124 if (data.cols() != input_dim_)
125 throw std::invalid_argument("Data dimension mismatch");
126
127 int n = static_cast<int>(data.rows());
128 if (batch_size <= 0 || batch_size > n) batch_size = n;
129
130 std::vector<int> indices(n);
131 std::iota(indices.begin(), indices.end(), 0);
132
133 std::vector<VAEEpochResult> history;
134
135 for (int epoch = 0; epoch < epochs; ++epoch) {
136 std::shuffle(indices.begin(), indices.end(), rng_);
137
138 double epoch_recon = 0, epoch_kl = 0;
139 int num_batches = 0;
140
141 for (int start = 0; start < n; start += batch_size) {
142 int end = std::min(start + batch_size, n);
143 int bs = end - start;
144
145 // Extract batch
146 MatrixD batch(bs, input_dim_);
147 for (int i = 0; i < bs; ++i)
148 batch.row(i) = data.row(indices[start + i]);
149
150 // ── E-step: forward pass ──
151 ForwardCache cache = forward(batch);
152
153 // ── Compute losses ──
154 double recon_loss = reconstruction_loss(batch, cache.x_hat);
155 double kl_loss = kl_divergence(cache.mu, cache.logvar);
156
157 epoch_recon += recon_loss * bs;
158 epoch_kl += kl_loss * bs;
159
160 // ── M-step: backward pass + parameter update ──
161 backward(cache);
162
163 ++num_batches;
164 }
165
166 VAEEpochResult result;
167 result.epoch = epoch;
168 result.reconstruction_loss = epoch_recon / n;
169 result.kl_divergence = epoch_kl / n;
170 result.total_loss = result.reconstruction_loss + result.kl_divergence;
171 history.push_back(result);
172 }
173
174 return history;
175 }

261 void init_weights() {
262 auto xavier = [&](int fan_in, int fan_out) -> MatrixD {
263 double scale = std::sqrt(2.0 / (fan_in + fan_out));
264 std::normal_distribution<double> dist(0.0, scale);
265 MatrixD W(fan_in, fan_out);
266 for (int i = 0; i < fan_in; ++i)
267 for (int j = 0; j < fan_out; ++j)
268 W(i, j) = dist(rng_);
269 return W;
270 };

306
307 ForwardCache forward(const MatrixD& x) {
308 ForwardCache c;
309 c.x = x;
310 int bs = static_cast<int>(x.rows());
311
312 // Encoder
313 c.h1 = (x * W1_).rowwise() + b1_.transpose();
314 c.h1_act = relu(c.h1);
315 c.mu = (c.h1_act * W_mu_).rowwise() + b_mu_.transpose();
316 c.logvar = (c.h1_act * W_logvar_).rowwise() + b_logvar_.transpose();
317
318 // Reparameterisation: z = μ + σ ⊙ ε
319 c.std = (0.5 * c.logvar.array()).exp().matrix();
320 c.eps = sample_noise(bs, latent_dim_);
321 c.z = c.mu + c.std.cwiseProduct(c.eps);
322
323 // Decoder
324 c.h2 = (c.z * W2_).rowwise() + b2_.transpose();
325 c.h2_act = relu(c.h2);
326 MatrixD logits = (c.h2_act * W3_).rowwise() + b3_.transpose();
327 c.x_hat = sigmoid(logits);
328
329 return c;
330 }

337 static double reconstruction_loss(const MatrixD& x, const MatrixD& x_hat) {
338 // Clamp x_hat to avoid log(0)
339 MatrixD xh = x_hat.array().max(1e-8).min(1.0 - 1e-8).matrix();
340 double bce = -(x.array() * xh.array().log()
341 + (1.0 - x.array()) * (1.0 - xh.array()).log()).sum();
342 return bce / x.rows();
343 }
344
346 static double kl_divergence(const MatrixD& mu, const MatrixD& logvar) {
347 // KL = -0.5 * Σ(1 + log σ² - μ² - σ²)
348 double kl = -0.5 * (1.0 + logvar.array() - mu.array().square()
349 - logvar.array().exp()).sum();
350 return kl / mu.rows();
351 }

353 // ---------------------------------------------------------------
354 // Backward pass (manual gradients)
355 // ---------------------------------------------------------------
356
357 void backward(const ForwardCache& c) {
358 int bs = static_cast<int>(c.x.rows());
359 double inv_bs = 1.0 / bs;
360
361 // ── Decoder gradients ──
362
363 // d_loss / d_logits = x_hat - x (BCE gradient through sigmoid)
364 MatrixD d_logits = (c.x_hat - c.x); // bs × input_dim
365
366 // Gradients for W3, b3
367 MatrixD dW3 = c.h2_act.transpose() * d_logits * inv_bs;
368 VectorD db3 = d_logits.colwise().mean().transpose();
369
370 // Backprop through relu
371 MatrixD d_h2_act = d_logits * W3_.transpose();
372 MatrixD d_h2 = d_h2_act.cwiseProduct(relu_grad(c.h2));
373
374 // Gradients for W2, b2
375 MatrixD dW2 = c.z.transpose() * d_h2 * inv_bs;
376 VectorD db2 = d_h2.colwise().mean().transpose();
377
378 // ── Latent gradients ──
379
380 // d_loss / d_z (from reconstruction)
381 MatrixD d_z = d_h2 * W2_.transpose();
382
383 // ── KL gradients (added to encoder gradients) ──
384 // d_KL / d_mu = mu
385 // d_KL / d_logvar = 0.5 * (exp(logvar) - 1)
386 MatrixD d_mu_kl = c.mu * inv_bs;
387 MatrixD d_logvar_kl = 0.5 * (c.logvar.array().exp() - 1.0).matrix() * inv_bs;
388
389 // ── Encoder gradients (through reparameterisation) ──
390
391 // d_z / d_mu = I
392 // d_z / d_std = eps
393 // d_z / d_logvar = 0.5 * std * eps (chain through std = exp(0.5*logvar))
394 MatrixD d_mu = d_z + d_mu_kl;
395 MatrixD d_logvar = d_z.cwiseProduct(c.eps).cwiseProduct(c.std) * 0.5 + d_logvar_kl;
396
397 // Gradients for W_mu, b_mu
398 MatrixD dW_mu = c.h1_act.transpose() * d_mu * inv_bs;
399 VectorD db_mu = d_mu.colwise().mean().transpose();
400
401 // Gradients for W_logvar, b_logvar
402 MatrixD dW_logvar = c.h1_act.transpose() * d_logvar * inv_bs;
403 VectorD db_logvar = d_logvar.colwise().mean().transpose();
404
405 // Backprop through encoder hidden
406 MatrixD d_h1_act = d_mu * W_mu_.transpose()
407 + d_logvar * W_logvar_.transpose();
408 MatrixD d_h1 = d_h1_act.cwiseProduct(relu_grad(c.h1));
409
410 // Gradients for W1, b1
411 MatrixD dW1 = c.x.transpose() * d_h1 * inv_bs;
412 VectorD db1 = d_h1.colwise().mean().transpose();
413
414 // ── M-step: gradient descent update ──
415 W1_ -= learning_rate_ * dW1;
416 b1_ -= learning_rate_ * db1;
417 W_mu_ -= learning_rate_ * dW_mu;
418 b_mu_ -= learning_rate_ * db_mu;
419 W_logvar_ -= learning_rate_ * dW_logvar;
420 b_logvar_ -= learning_rate_ * db_logvar;
421 W2_ -= learning_rate_ * dW2;
422 b2_ -= learning_rate_ * db2;
423 W3_ -= learning_rate_ * dW3;
424 b3_ -= learning_rate_ * db3;
425 }
