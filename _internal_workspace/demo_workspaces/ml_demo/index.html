Example usage
Python kmeans_example.py 
from ml_toolbox import kmeans
import numpy as np

# Generate 2D sample data (3 clusters)
rng = np.random.default_rng(42)
X = np.vstack([
    rng.normal(loc=[0.2, 0.2], scale=0.07, size=(100, 2)),
    rng.normal(loc=[0.7, 0.3], scale=0.07, size=(100, 2)),
    rng.normal(loc=[0.5, 0.8], scale=0.07, size=(100, 2)),
])

# Fit K-Means with k=3 clusters
result = kmeans.fit(X, k=3)

print("Centroids:\n", result.centroids)
print("Labels:    ", result.labels[:10], "...")
print("Inertia:   ", round(result.inertia, 4))
print("Iterations:", result.n_iter)
