36enum class ExprType {
37 LITERAL, // 42, 'hello', TRUE, NULL
38 IDENTIFIER, // column or table.column
39 BINARY_OP, // left OP right
40 UNARY_OP, // NOT expr, -expr
41 FUNCTION_CALL, // COUNT(*), MAX(x)
42 STAR, // *
43 PLACEHOLDER, // ?, :name, $1
44 BETWEEN, // expr BETWEEN low AND high
45 IN_LIST, // expr IN (v1, v2, …)
46 IS_NULL, // expr IS [NOT] NULL
47 SUBQUERY, // (SELECT …) — placeholder for nested selects
48};

50struct Expression {
51 ExprType type;

53 // LITERAL / IDENTIFIER / STAR / PLACEHOLDER

59 // Children
60 ExprPtr left;

62
63 // FUNCTION_CALL arguments / IN_LIST values
64 std::vector<ExprPtr> args;
65
66 // BETWEEN low/high (reuses left=low, right=high)
67
68 // IS_NULL negated?
69 bool negated = false;
70
71 // Optional alias (AS …)
72 std::string alias;
