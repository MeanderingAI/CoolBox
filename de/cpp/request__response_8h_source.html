79 // Parse a raw HTTP request string into a Request object
80 static Request from_string(const std::string& buffer) {
81 Request req;
82 size_t method_end = buffer.find(' ');
83 if (method_end != std::string::npos) {
84 req.method = buffer.substr(0, method_end);
85 size_t path_start = method_end + 1;
86 size_t path_end = buffer.find(' ', path_start);
87 if (path_end != std::string::npos) {
88 req.uri = buffer.substr(path_start, path_end - path_start);
89 }
90 }
91 // Parse headers and body
92 size_t header_start = buffer.find("\r\n") + 2;
93 size_t body_pos = buffer.find("\r\n\r\n");
94 if (body_pos != std::string::npos) {
95 // Parse headers
96 size_t pos = header_start;
97 while (pos < body_pos) {
98 size_t line_end = buffer.find("\r\n", pos);
99 if (line_end == std::string::npos || line_end > body_pos) break;
100 std::string line = buffer.substr(pos, line_end - pos);
101 size_t colon = line.find(":");
102 if (colon != std::string::npos) {
103 std::string key = line.substr(0, colon);
104 std::string value = line.substr(colon + 1);
105 // Trim whitespace
106 key.erase(0, key.find_first_not_of(" \t"));
107 key.erase(key.find_last_not_of(" \t") + 1);
108 value.erase(0, value.find_first_not_of(" \t"));
109 value.erase(value.find_last_not_of(" \t") + 1);
110 HeaderKey hkey = header_key_from_string(key);
111 if (hkey == HeaderKey::Custom)
112 req.headers[key] = value;
113 else
114 req.headers[hkey] = value;
115 }
116 pos = line_end + 2;
117 }
118 req.body = buffer.substr(body_pos + 4);
119 }
120 return req;
121 }
