229 std::vector<double> query(int query_node,
230 const std::map<int, int>& evidence = {}) const {
231 validate_id(query_node);
232
233 // Build cardinality map
234 std::map<int, int> all_cards;
235 for (const auto& n : nodes_)
236 all_cards[n.id] = static_cast<int>(n.states.size());
237
238 // Create initial factors from CPTs
239 std::vector<Factor> factors;
240 for (const auto& node : nodes_) {
241 if (node.cpt.empty())
242 throw std::runtime_error("CPT not set for node '" + node.name + "'");
243 factors.push_back(make_factor(node));
244 }
245
246 // Apply evidence: reduce factors
247 for (const auto& [var, val] : evidence) {
248 for (auto& f : factors) {
249 if (std::find(f.variables.begin(), f.variables.end(), var)
250 != f.variables.end()) {
251 f = f.reduce(var, val);
252 }
253 }
254 }
255
256 // Variable elimination order: eliminate all non-query, non-evidence vars
257 std::vector<int> elim_order;
258 for (const auto& n : nodes_) {
259 if (n.id != query_node && evidence.find(n.id) == evidence.end())
260 elim_order.push_back(n.id);
261 }
262
263 // Eliminate variables one by one
264 for (int var : elim_order) {
265 // Collect factors that mention this variable
266 std::vector<Factor> relevant, remaining;
267 for (auto& f : factors) {
268 if (std::find(f.variables.begin(), f.variables.end(), var)
269 != f.variables.end()) {
270 relevant.push_back(f);
271 } else {
272 remaining.push_back(f);
273 }
274 }
275
276 if (relevant.empty()) {
277 factors = remaining;
278 continue;
279 }
280
281 // Multiply all relevant factors
282 Factor product = relevant[0];
283 for (size_t i = 1; i < relevant.size(); ++i) {
284 product = Factor::multiply(product, relevant[i], all_cards);
285 }
286
287 // Marginalise out the variable
288 Factor marginalised = product.marginalise(var);
289 remaining.push_back(marginalised);
290 factors = remaining;
291 }
292
293 // Multiply remaining factors
294 Factor result = factors[0];
295 for (size_t i = 1; i < factors.size(); ++i) {
296 result = Factor::multiply(result, factors[i], all_cards);
297 }
298
299 // Normalise
300 result.normalise();
301 return result.table;
302 }

333 std::vector<int> topological_order() const {
334 std::vector<int> order;
335 std::set<int> visited;
336 std::set<int> in_stack;
337
338 std::function<void(int)> dfs = [&](int id) {
339 if (in_stack.count(id))
340 throw std::runtime_error("Cycle detected in network");
341 if (visited.count(id)) return;
342 in_stack.insert(id);
343 for (int child : nodes_[id].children)
344 dfs(child);
345 in_stack.erase(id);
346 visited.insert(id);
347 order.push_back(id);
348 };
349
350 for (const auto& n : nodes_)
351 if (!visited.count(n.id))
352 dfs(n.id);
353
354 std::reverse(order.begin(), order.end());
355 return order;
356 }

426 bool has_path(int from, int to) const {
427 if (from == to) return true;
428 std::set<int> visited;
429 std::vector<int> stack = {from};
430 while (!stack.empty()) {
431 int curr = stack.back();
432 stack.pop_back();
433 if (curr == to) return true;
434 if (visited.count(curr)) continue;
435 visited.insert(curr);
436 for (int child : nodes_[curr].children)
437 stack.push_back(child);
438 }
439 return false;
440 }

464inline Factor Factor::multiply(const Factor& f1, const Factor& f2,
465 const std::map<int, int>& all_cards) {
466 // Union of variables
467 std::vector<int> new_vars = f1.variables;
468 for (int v : f2.variables) {
469 if (std::find(new_vars.begin(), new_vars.end(), v) == new_vars.end())
470 new_vars.push_back(v);
471 }
472
473 std::vector<int> new_card;
474 for (int v : new_vars)
475 new_card.push_back(all_cards.at(v));
476
477 Factor result(new_vars, new_card, {});
478 result.table.resize(result.size(), 0.0);
479
480 // Build index maps: for each var in new_vars, find its position in f1 and f2
481 auto pos_in = [](const std::vector<int>& vars, int v) -> int {
482 auto it = std::find(vars.begin(), vars.end(), v);
483 return (it != vars.end()) ? static_cast<int>(std::distance(vars.begin(), it)) : -1;
484 };
485
486 for (int i = 0; i < result.size(); ++i) {
487 std::vector<int> assignment = result.index_to_assignment(i);
488
489 // Map to f1 assignment
490 std::vector<int> a1(f1.variables.size());
491 for (size_t j = 0; j < f1.variables.size(); ++j) {
492 int p = pos_in(new_vars, f1.variables[j]);
493 a1[j] = assignment[p];
494 }
495
496 // Map to f2 assignment
497 std::vector<int> a2(f2.variables.size());
498 for (size_t j = 0; j < f2.variables.size(); ++j) {
499 int p = pos_in(new_vars, f2.variables[j]);
500 a2[j] = assignment[p];
501 }
502
503 result.table[i] = f1.table[f1.assignment_to_index(a1)]
504 * f2.table[f2.assignment_to_index(a2)];
505 }
506
507 return result;
508}

510inline Factor Factor::marginalise(int var) const {
511 auto it = std::find(variables.begin(), variables.end(), var);
512 if (it == variables.end()) return *this;
513
514 int pos = static_cast<int>(std::distance(variables.begin(), it));
515 int var_card = card[pos];
516
517 std::vector<int> new_vars, new_card;
518 for (size_t i = 0; i < variables.size(); ++i) {
519 if (static_cast<int>(i) != pos) {
520 new_vars.push_back(variables[i]);
521 new_card.push_back(card[i]);
522 }
523 }
524
525 Factor result(new_vars, new_card, {});
526 result.table.resize(result.size(), 0.0);
527
528 for (int i = 0; i < this->size(); ++i) {
529 std::vector<int> assignment = index_to_assignment(i);
530
531 // Build reduced assignment (skip the marginalised variable)
532 std::vector<int> reduced;
533 for (size_t j = 0; j < assignment.size(); ++j) {
534 if (static_cast<int>(j) != pos)
535 reduced.push_back(assignment[j]);
536 }
537
538 if (!reduced.empty()) {
539 int new_idx = result.assignment_to_index(reduced);
540 result.table[new_idx] += table[i];
541 } else {
542 // All variables marginalised — single scalar
543 result.table[0] += table[i];
544 }
545 }
546
547 return result;
548}

550inline Factor Factor::reduce(int var, int value) const {
551 auto it = std::find(variables.begin(), variables.end(), var);
552 if (it == variables.end()) return *this;
553
554 int pos = static_cast<int>(std::distance(variables.begin(), it));
555
556 std::vector<int> new_vars, new_card;
557 for (size_t i = 0; i < variables.size(); ++i) {
558 if (static_cast<int>(i) != pos) {
559 new_vars.push_back(variables[i]);
560 new_card.push_back(card[i]);
561 }
562 }
563
564 Factor result(new_vars, new_card, {});
565 result.table.resize(result.size(), 0.0);
566
567 for (int i = 0; i < this->size(); ++i) {
568 std::vector<int> assignment = index_to_assignment(i);
569 if (assignment[pos] != value) continue;
570
571 std::vector<int> reduced;
572 for (size_t j = 0; j < assignment.size(); ++j) {
573 if (static_cast<int>(j) != pos)
574 reduced.push_back(assignment[j]);
575 }
576
577 if (!reduced.empty()) {
578 result.table[result.assignment_to_index(reduced)] = table[i];
579 } else {
580 result.table[0] = table[i];
581 }
582 }
583
584 return result;
