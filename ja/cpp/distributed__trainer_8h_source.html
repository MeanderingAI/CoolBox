11// Strategy for distributed training
12enum class TrainingStrategy {
13 DATA_PARALLEL, // Each worker has full model, different data
14 MODEL_PARALLEL, // Model split across workers
15 PARAMETER_SERVER, // Centralized parameter management
16 DECENTRALIZED, // Peer-to-peer gradient exchange
17 FEDERATED // Federated learning (local training + aggregation)
18};

20// Gradient aggregation method
21enum class AggregationMethod {
22 SYNCHRONOUS, // Wait for all workers
23 ASYNCHRONOUS, // Update as gradients arrive
