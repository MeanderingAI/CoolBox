159 void fit(const MatrixT& data, int epochs = 100) {
160 if (data.cols() != input_dim_)
161 throw std::invalid_argument("Data dimension mismatch");
162
163 int n_samples = static_cast<int>(data.rows());
164 int total_iterations = epochs * n_samples;
165 time_constant_ = static_cast<Scalar>(total_iterations)
166 / std::log(initial_radius_ + 1);
167
168 // Indices for shuffling
169 std::vector<int> indices(n_samples);
170 std::iota(indices.begin(), indices.end(), 0);
171
172 int iteration = 0;
173 for (int epoch = 0; epoch < epochs; ++epoch) {
174 std::shuffle(indices.begin(), indices.end(), rng_);
175
176 for (int idx : indices) {
177 VectorT sample = data.row(idx).transpose();
178
179 // 1. Find BMU
180 BMUResult bmu = find_bmu(sample);
181
182 // 2. Compute decayed parameters
183 Scalar lr = learning_rate(iteration, total_iterations);
184 Scalar radius = neighbourhood_radius(iteration, total_iterations);
185
186 // 3. Update weights
187 update_weights(sample, bmu, lr, radius);
188
189 ++iteration;
190 }
191 }
192 }

211 BMUResult find_bmu(const VectorT& sample) const {
212 BMUResult result{0, 0, 0, std::numeric_limits<Scalar>::max()};
213
214 for (int i = 0; i < rows_ * cols_; ++i) {
215 Scalar dist = (weights_.row(i).transpose() - sample).squaredNorm();
216 if (dist < result.distance) {
217 result.flat_index = i;
218 result.row = i / cols_;
219 result.col = i % cols_;
220 result.distance = dist;
221 }
222 }
223 result.distance = std::sqrt(result.distance);
224 return result;
225 }

268 MatrixT u_matrix() const {
269 MatrixT umat = MatrixT::Zero(rows_, cols_);
270
271 for (int r = 0; r < rows_; ++r) {
272 for (int c = 0; c < cols_; ++c) {
273 VectorT w = weights_.row(r * cols_ + c).transpose();
274 Scalar sum = 0;
275 int count = 0;
276
277 // Check 4-connected neighbours
278 int dr[] = {-1, 1, 0, 0};
279 int dc[] = {0, 0, -1, 1};
280 for (int d = 0; d < 4; ++d) {
281 int nr = r + dr[d];
282 int nc = c + dc[d];
283 if (nr >= 0 && nr < rows_ && nc >= 0 && nc < cols_) {
284 VectorT nw = weights_.row(nr * cols_ + nc).transpose();
285 sum += (w - nw).norm();
286 ++count;
287 }
288 }
289 umat(r, c) = (count > 0) ? sum / count : 0;
290 }
291 }
292 return umat;

299 Scalar topographic_error(const MatrixT& data) const {
300 int errors = 0;
301 for (int i = 0; i < data.rows(); ++i) {
302 VectorT sample = data.row(i).transpose();
303
304 // Find top-2 BMUs
305 int best = -1, second = -1;
306 Scalar best_dist = std::numeric_limits<Scalar>::max();
307 Scalar second_dist = std::numeric_limits<Scalar>::max();
308
309 for (int j = 0; j < rows_ * cols_; ++j) {
310 Scalar d = (weights_.row(j).transpose() - sample).squaredNorm();
311 if (d < best_dist) {
312 second = best; second_dist = best_dist;
313 best = j; best_dist = d;
314 } else if (d < second_dist) {
315 second = j; second_dist = d;
316 }
317 }
318
319 // Check adjacency
320 int r1 = best / cols_, c1 = best % cols_;
321 int r2 = second / cols_, c2 = second % cols_;
322 int manhattan = std::abs(r1 - r2) + std::abs(c1 - c2);
323 if (manhattan > 1) ++errors;
324 }
325 return static_cast<Scalar>(errors) / static_cast<Scalar>(data.rows());
326 }

366 Scalar neighbourhood_influence(Scalar grid_dist, Scalar radius) const {
367 if (radius <= 0) return 0;
368 switch (neighbourhood_type_) {
369 case NeighbourhoodType::GAUSSIAN:
370 return std::exp(-(grid_dist * grid_dist) /
371 (2.0 * radius * radius));
372 case NeighbourhoodType::BUBBLE:
373 return (grid_dist <= radius) ? 1.0 : 0.0;
374 default:
375 return 0;
376 }

379 // ---------------------------------------------------------------
380 // Weight update

382
383 void update_weights(const VectorT& sample, const BMUResult& bmu,
384 Scalar lr, Scalar radius) {
385 for (int r = 0; r < rows_; ++r) {
386 for (int c = 0; c < cols_; ++c) {
387 Scalar dr = static_cast<Scalar>(r - bmu.row);
388 Scalar dc = static_cast<Scalar>(c - bmu.col);
389 Scalar grid_dist = std::sqrt(dr * dr + dc * dc);
390
391 Scalar influence = neighbourhood_influence(grid_dist, radius);
392 if (influence < 1e-10) continue;
393
394 int idx = r * cols_ + c;
395 weights_.row(idx) += lr * influence *
396 (sample.transpose() - weights_.row(idx));
397 }
398 }

405 void init_random_uniform(const MatrixT& data) {
406 VectorT mins = data.colwise().minCoeff().transpose();
407 VectorT maxs = data.colwise().maxCoeff().transpose();
408
409 std::uniform_real_distribution<Scalar> dist(0.0, 1.0);
410 for (int i = 0; i < rows_ * cols_; ++i) {
411 for (int d = 0; d < input_dim_; ++d) {
412 weights_(i, d) = mins(d) + dist(rng_) * (maxs(d) - mins(d));
413 }
414 }
415 }

417 void init_random_sample(const MatrixT& data) {
418 std::uniform_int_distribution<int> dist(0, static_cast<int>(data.rows()) - 1);
419 for (int i = 0; i < rows_ * cols_; ++i) {
420 weights_.row(i) = data.row(dist(rng_));
421 }
422 }

424 void init_pca(const MatrixT& data) {
425 // Centre data
426 VectorT mean = data.colwise().mean().transpose();
427 MatrixT centred = data.rowwise() - mean.transpose();
428
429 // Compute covariance and top-2 eigenvectors
430 MatrixT cov = (centred.transpose() * centred) /
431 static_cast<Scalar>(data.rows() - 1);
432 Eigen::SelfAdjointEigenSolver<MatrixT> solver(cov);
433 MatrixT eigvecs = solver.eigenvectors();
434
435 // Take the two largest eigenvectors
436 VectorT pc1 = eigvecs.col(input_dim_ - 1);
437 VectorT pc2 = (input_dim_ >= 2) ? VectorT(eigvecs.col(input_dim_ - 2))
438 : VectorT::Zero(input_dim_);
439
440 // Spread weights along PC1 and PC2
441 for (int r = 0; r < rows_; ++r) {
442 for (int c = 0; c < cols_; ++c) {
443 Scalar t1 = (rows_ > 1)
444 ? (static_cast<Scalar>(r) / (rows_ - 1) - 0.5) * 2
445 : 0;
446 Scalar t2 = (cols_ > 1)
447 ? (static_cast<Scalar>(c) / (cols_ - 1) - 0.5) * 2
448 : 0;
449 VectorT w = mean + t1 * pc1 + t2 * pc2;
450 weights_.row(r * cols_ + c) = w.transpose();
451 }
452 }
453 }
