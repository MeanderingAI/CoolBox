132 double pack_internal_resistance() const {
133 // Series: R adds per stage
134 // Parallel: 1/R_total = sum(1/R_i)
135 double r_total = 0.0;
136 for (int s = 0; s < series_count_; ++s) {
137 double g_stage = 0.0; // conductance sum for parallel
138 for (int p = 0; p < parallel_count_; ++p) {
139 double r = cells_[idx(s, p)].internal_resistance();
140 if (r > 0) g_stage += 1.0 / r;
141 }
142 r_total += (g_stage > 0) ? (1.0 / g_stage) : 0.0;
143 }
144 return r_total;
145 }
