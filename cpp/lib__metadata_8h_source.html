13// Platform-specific section attribute for embedding metadata
14#if defined(_MSC_VER)
15 // MSVC: use __declspec(allocate) with a pragma section, or simply skip section embedding
16 #define LIB_META_SECTION
17 #define LIB_META_EXPORT __declspec(dllexport)
18#elif defined(__APPLE__)
19 #define LIB_META_SECTION __attribute__((used, section("__DATA,__lib_meta")))
20 #define LIB_META_EXPORT __attribute__((visibility("default")))
21#else
22 #define LIB_META_SECTION __attribute__((used, section(".lib_meta")))
23 #define LIB_META_EXPORT __attribute__((visibility("default")))
24#endif

26// Macro to embed library metadata and Doxygen doc block with unique function names
27#define LIBRARY_METADATA_DOXYGEN(libid, name, version, description, author) \
28 extern "C" { \
29 LIB_META_SECTION \
30 static const char _lib_##libid##_name[] = name; \
31 LIB_META_SECTION \
32 static const char _lib_##libid##_version[] = version; \
33 LIB_META_SECTION \
34 static const char _lib_##libid##_description[] = description; \
35 LIB_META_SECTION \
36 static const char _lib_##libid##_author[] = author; \
37 LIB_META_EXPORT \
38 const char* get_##libid##_library_name() { return name; } \
39 LIB_META_EXPORT \
40 const char* get_##libid##_library_version() { return version; } \
41 LIB_META_EXPORT \
42 const char* get_##libid##_library_description() { return description; } \
43 LIB_META_EXPORT \
44 const char* get_##libid##_library_author() { return author; } \
45 }
