54 // Run a shell command and return output lines
55 static std::vector<std::string> run(const std::string& cmd) {
56 std::vector<std::string> result;
57 FILE* pipe = popen(cmd.c_str(), "r");
58 if (!pipe) throw std::runtime_error("popen() failed!");
59 char buffer[1024];
60 while (fgets(buffer, sizeof(buffer), pipe)) {
61 std::string line(buffer);
62 line.erase(line.find_last_not_of(" \n\r\t") + 1);
63 result.push_back(line);
64 }
65 pclose(pipe);
66 return result;
67 }
