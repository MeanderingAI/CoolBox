14// Produces a collection of drawing primitives (circles + lines + text)
15// representing atoms and bonds. These primitives are renderer-agnostic
16// and can be painted into a canvas implementation.
17
18inline std::vector<Primitive> render_atom_bond_primitives(
19 const std::vector<std::string>& atoms,
20 const std::vector<std::pair<int,int>>& bonds,
21 float start_x = 10.0f,
22 float start_y = 50.0f,
23 float spacing = 48.0f)
24{
25 std::vector<Primitive> out;
26 size_t n = atoms.size();
27 if (n == 0) return out;
28
29 // simple linear layout: atom i at (start_x + i*spacing, start_y)
30 std::vector<std::pair<float,float>> pos(n);
31 for (size_t i = 0; i < n; ++i) {
32 pos[i] = { start_x + static_cast<float>(i) * spacing, start_y };
33 }
34
35 // Draw bonds as lines between centers
36 for (auto &b : bonds) {
37 int a = b.first;
38 int c = b.second;
39 if (a < 0 || a >= static_cast<int>(n) || c < 0 || c >= static_cast<int>(n)) continue;
40 Line L{ pos[a].first, pos[a].second, pos[c].first, pos[c].second, 2.0f, rgba(80,80,80) };
41 out.emplace_back(L);
42 }
43
44 // Draw atoms as circles with text (symbol + valence)
45 for (size_t i = 0; i < n; ++i) {
46 auto [x,y] = pos[i];
47 Circle C{ x, y, 16.0f, rgba(200,220,255) };
48 out.emplace_back(C);
49
50 auto elem = chemistry::element_by_symbol(atoms[i]);
51 std::ostringstream label;
52 label << elem.symbol;
53 if (elem.valence_electrons > 0) label << " (v=" << elem.valence_electrons << ")";
54 Text T{ x - 12.0f, y - 6.0f, label.str(), 12, rgba(10,10,10) };
55 out.emplace_back(T);
56 }
57
58 return out;
59}
