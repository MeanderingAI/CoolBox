Example: Plotting with the Graphics Library

import ml_core.graphics

graph = ml_core.graphics.Graph(400, 300, ml_core.graphics.GraphType.LINE)
graph.set_title("Line Plot Example")
graph.set_x_label("X")
graph.set_y_label("Y")
series = ml_core.graphics.DataSeries(
	"Demo Series", [0, 1, 2, 3], [1, 3, 2, 4], ml_core.graphics.BLUE
)
graph.add_series(series)
canvas = graph.render()
canvas.save("coolbox_line_plot.png")
print("Plot saved as coolbox_line_plot.png")

Example: Wave Generation and Plotting

import ml_core.graphics
import ml_core.misc.wave_generator
import numpy as np

# Configure wave
config = ml_core.misc.wave_generator.WaveConfig()
config.amplitude = 1.0
config.frequency_hz = 2.0

# Generate samples
duration = 1.0  # seconds
rate = 100      # samples per second
samples = ml_core.misc.wave_generator.generate_samples_for_duration(
	duration, rate, ml_core.misc.wave_generator.WavePattern.SINE, config
)
x = np.linspace(0, duration, len(samples))

# Plot wave
graph = ml_core.graphics.Graph(400, 200, ml_core.graphics.GraphType.LINE)
graph.set_title("Sine Wave")
graph.set_x_label("Time (s)")
graph.set_y_label("Amplitude")
series = ml_core.graphics.DataSeries("Sine", x.tolist(), samples, ml_core.graphics.GREEN)
graph.add_series(series)
canvas = graph.render()
canvas.save("coolbox_sine_wave.png")
print("Wave plot saved as coolbox_sine_wave.png")

Example: Linear Regression

from coolbox import LinearRegression
import numpy as np

# Generate synthetic data
X = np.array([[1], [2], [3], [4], [5]])
y = np.array([2, 4, 6, 8, 10])

# Fit model
model = LinearRegression()
model.fit(X, y)

# Predict
pred = model.predict(np.array([[6], [7]]))
print("Predictions:", pred)

Example: Decision Tree Classifier

from coolbox import DecisionTreeClassifier
import numpy as np

X = np.array([[0, 0], [1, 1], [0, 1], [1, 0]])
y = np.array([0, 1, 1, 0])

clf = DecisionTreeClassifier()
clf.fit(X, y)
print("Prediction for [1, 1]:", clf.predict(np.array([[1, 1]])))
